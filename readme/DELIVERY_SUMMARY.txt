================================================================================
                         USAGE EXAMPLES
================================================================================

EXAMPLE 1: ML BACKTEST WITH RISK
from strategies.advanced import MachineLearningStrategy
from portfolio.risk_manager import RiskManager
from backtesting.backtest_engine import BacktestEngine

strategy = MachineLearningStrategy()
risk_mgr = RiskManager(max_daily_loss=0.05)
engine = BacktestEngine(100000, strategy, risk_mgr)
results = engine.run(data, 'AAPL')
print(f"Return: {results['total_return']:.2%}")

EXAMPLE 2: SAVE & QUERY BACKTESTS
from utils.database import TradingDatabase

db = TradingDatabase('trading.db')
id = db.save_backtest('ML Test', '2023-01-01', '2023-12-31', 
                      ['AAPL'], 100000, 'machine_learning', 
                      {'lookback': 50}, results)
backtests = db.get_backtests(limit=50)

EXAMPLE 3: CREATE PRICE ALERTS
from utils.alerts import AlertManager, RealTimeMonitor

alerts = AlertManager()
monitor = RealTimeMonitor(alerts)
monitor.add_price_target('AAPL', 160.0, 'above')
monitor.check_price_targets('AAPL', 159.99)

EXAMPLE 4: ENFORCE RISK LIMITS
from portfolio.risk_manager import RiskManager

risk = RiskManager()
max_size = risk.get_max_trade_size(portfolio_value)
stop_price = risk.calculate_position_stop_loss(entry_price)
ok = risk.check_position_size(portfolio_value, position_size)
