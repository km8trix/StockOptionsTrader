================================================================================
3 WAYS TO USE IT
================================================================================

WAY 1: BACKTEST EXISTING STRATEGIES
  from backtesting.backtest_engine import BacktestEngine
  from strategies.base import MomentumStrategy
  
  strategy = MomentumStrategy()
  backtester = BacktestEngine(strategy)
  results = backtester.run(['AAPL', 'MSFT'], '2023-01-01', '2024-01-01')
  print(results['summary'])

WAY 2: WRITE & TEST CUSTOM STRATEGIES
  from strategies.base import Strategy
  
  class MyStrategy(Strategy):
      def generate_signals(self, data, asset):
          # Your logic here
          if data.iloc[-1]['close'] > data.iloc[-1]['sma_20']:
              return 'BUY'
          return 'HOLD'
  
  # Backtest it
  backtester = BacktestEngine(MyStrategy())
  results = backtester.run(['AAPL'], '2023-01-01', '2024-01-01')

WAY 3: PAPER TRADE WITH SIMULATED ORDERS
  from brokers.paper_trader import PaperTrader
  from core.models import Asset, AssetType, OrderType
  
  trader = PaperTrader(50000)
  apple = Asset('AAPL', AssetType.STOCK)
  trader.place_order(apple, OrderType.BUY, 100, 150.00)
  status = trader.get_portfolio_status()
